Shader "Custom/StaticLightmappedGlossMasked" {
Properties {
 _MainTex ("Base (RGB)", 2D) = "white" {}
 _GlossTex ("Gloss (G)", 2D) = "white" {}
 _SpecColor ("Specular Colour", Color) = (1,1,1,1)
 _Shininess2 ("Specular Power", Range(0,128)) = 10
}
SubShader { 
 Pass {
Program "vp" {
SubProgram "gles " {
"!!GLES

#define SHADER_API_GLES 1
#define tex2D texture2D
#line 21

			
			// interpolators
			varying mediump vec2 i_mainUV;
			varying mediump vec2 i_lightmapUV;
			varying lowp vec3 i_specular;
			
		  			
		  			
			
#ifdef VERTEX
#define gl_ModelViewProjectionMatrix glstate_matrix_mvp
uniform mat4 glstate_matrix_mvp;
#define gl_Vertex _glesVertex
attribute vec4 _glesVertex;
#define gl_Normal _glesNormal
attribute vec3 _glesNormal;
#define gl_MultiTexCoord0 _glesMultiTexCoord0
attribute vec4 _glesMultiTexCoord0;
#define gl_MultiTexCoord1 _glesMultiTexCoord1
attribute vec4 _glesMultiTexCoord1;
   
   // custom uniform variables
   uniform highp float _Shininess2;
   uniform lowp vec3 _SpecColor;
   uniform highp vec3 g_highlightDirection;
      
   // unity variables
   uniform highp vec3 _WorldSpaceCameraPos;
   uniform highp vec4 unity_LightmapST;
   
   void main()
   {
    gl_Position = gl_ModelViewProjectionMatrix * vec4(gl_Vertex.xyz,1.0);
    i_mainUV = gl_MultiTexCoord0.xy;
    i_lightmapUV = gl_MultiTexCoord1.xy * unity_LightmapST.xy + unity_LightmapST.zw;
    
    highp vec3 viewDir = normalize(_WorldSpaceCameraPos - gl_Vertex.xyz);
    highp vec3 halfVector = normalize(viewDir - g_highlightDirection.xyz);
    highp float specDot = max(dot(halfVector, gl_Normal), 0.0);
    lowp float specular = pow(specDot, _Shininess2);
       i_specular = _SpecColor * specular;
   }
    
#endif
#ifdef FRAGMENT

      
// Decodes lightmaps:
// - doubleLDR encoded on GLES
// - RGBM encoded with range [0;8] on other platforms using surface shaders
lowp vec3 DecodeLightmapHalf( lowp vec4 color )
{
#if defined(SHADER_API_GLES)
 return color.rgb;
#else
 // potentially faster to do the scalar multiplication
 // in parenthesis for scalar GPUs
 return (4.0 * color.a) * color.rgb;
#endif
}

      
   uniform sampler2D _MainTex;
   uniform sampler2D _GlossTex;
   uniform sampler2D unity_Lightmap;
   
   void main()
   {
    // apply old Main Color term to both diffuse and specular, specular color adjusted by 255/150
    // apply to lightmap so it can be combined with the *2.0 term in DecodeLightmap
    lowp float c_mainColour = 2.0 * 150.0/255.0;
    lowp vec3 lightmap = c_mainColour * DecodeLightmapHalf(texture2D(unity_Lightmap, i_lightmapUV));
    lowp vec3 diffuse = texture2D(_MainTex, i_mainUV).rgb;
    lowp vec3 gloss = texture2D(_GlossTex, i_mainUV).rgb;
    lowp vec3 colour = (diffuse + i_specular*gloss) * lightmap;
    gl_FragData[0].xyz = colour;
    gl_FragData[0].w = 1.0;
   }
   
     
#endif"
}
}
 }
}

  gl_FragData[0] = tmpvar_1;
}



#endif"
